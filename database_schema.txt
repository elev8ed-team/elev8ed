-- =========================================================================
-- 1. ADJUSTED STATE ENUMS
-- =========================================================================

CREATE TYPE user_role AS ENUM ('super_core', 'department_head', 'subhead', 'executive');
CREATE TYPE member_status AS ENUM ('invited', 'active', 'left');
CREATE TYPE task_status AS ENUM ('todo', 'in_progress', 'done');
CREATE TYPE task_priority AS ENUM ('low', 'medium', 'high', 'urgent');
CREATE TYPE candidate_stage AS ENUM ('applied', 'written_round', 'interview_scheduled', 'evaluation', 'selected', 'rejected');

-- =========================================================================
-- 2. WORKSPACE INFRASTRUCTURE (With Subscription Ownership Mapping)
-- =========================================================================

CREATE TABLE workspaces (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name TEXT NOT NULL, -- Stored as two-tier display name: e.g., "IIT Bombay — Chemistry Club"
    slug TEXT UNIQUE NOT NULL, -- URL-safe identifier: e.g., "iit-bombay-chemistry-club"
    description TEXT,
    logo_url TEXT,
    owner_id UUID REFERENCES auth.users(id) ON DELETE SET NULL, -- References global Auth
    created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE tenures (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    workspace_id UUID REFERENCES workspaces(id) ON DELETE CASCADE,
    year_label TEXT NOT NULL, -- e.g., "2026-2027"
    is_active BOOLEAN DEFAULT true, -- True = Current active operations; False = Sealed historical archive
    created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE departments (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    workspace_id UUID REFERENCES workspaces(id) ON DELETE CASCADE,
    name TEXT NOT NULL,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- =========================================================================
-- 3. MEMBERS SYSTEM (Multi-Tenant Ready with Dedicated user_id Link)
-- =========================================================================

CREATE TABLE members (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(), -- Independent membership card ID
    user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE, -- Bridges to Google OAuth account
    workspace_id UUID REFERENCES workspaces(id) ON DELETE CASCADE,
    tenure_id UUID REFERENCES tenures(id) ON DELETE SET NULL,
    department_id UUID REFERENCES departments(id) ON DELETE SET NULL, -- NULL naturally for Super Core users
    full_name TEXT NOT NULL,
    role user_role NOT NULL DEFAULT 'executive',
    status member_status NOT NULL DEFAULT 'invited',
    joined_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(user_id, workspace_id, tenure_id) -- Prevents duplicate memberships in the same academic cycle
);

-- =========================================================================
-- 4. CROSS-DEPARTMENT TASK MATRIX
-- =========================================================================

CREATE TABLE tasks (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    workspace_id UUID REFERENCES workspaces(id) ON DELETE CASCADE,
    tenure_id UUID REFERENCES tenures(id) ON DELETE CASCADE,
    department_id UUID REFERENCES departments(id) ON DELETE CASCADE,
    title TEXT NOT NULL,
    description TEXT,
    status task_status NOT NULL DEFAULT 'todo',
    priority task_priority NOT NULL DEFAULT 'medium',
    assigned_to UUID REFERENCES members(id) ON DELETE SET NULL,
    created_by UUID REFERENCES members(id) ON DELETE SET NULL,
    deadline TIMESTAMPTZ,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- =========================================================================
-- 5. RECRUITMENT AND PANEL INTAKE PIPELINE
-- =========================================================================

CREATE TABLE candidates (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    workspace_id UUID REFERENCES workspaces(id) ON DELETE CASCADE,
    tenure_id UUID REFERENCES tenures(id) ON DELETE CASCADE,
    full_name TEXT NOT NULL,
    email TEXT NOT NULL,
    phone TEXT,
    preferred_dept_id UUID REFERENCES departments(id) ON DELETE SET NULL,
    stage candidate_stage NOT NULL DEFAULT 'applied',
    resume_url TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE candidate_reviews (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    candidate_id UUID REFERENCES candidates(id) ON DELETE CASCADE,
    reviewer_id UUID REFERENCES members(id) ON DELETE CASCADE,
    score_technical INT CHECK (score_technical BETWEEN 1 AND 10),
    score_communication INT CHECK (score_communication BETWEEN 1 AND 10),
    score_cultural INT CHECK (score_cultural BETWEEN 1 AND 10),
    comments TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- =========================================================================
-- 6. ONBOARDING & INVITATION MATRICES
-- =========================================================================

CREATE TABLE workspace_invitations (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    workspace_id UUID REFERENCES workspaces(id) ON DELETE CASCADE,
    tenure_id UUID REFERENCES tenures(id) ON DELETE CASCADE,
    department_id UUID REFERENCES departments(id) ON DELETE SET NULL,
    candidate_id UUID REFERENCES candidates(id) ON DELETE SET NULL,
    email TEXT NOT NULL,
    role user_role NOT NULL DEFAULT 'executive',
    token UUID NOT NULL DEFAULT gen_random_uuid() UNIQUE,
    invited_by UUID REFERENCES members(id) ON DELETE SET NULL,
    expires_at TIMESTAMPTZ NOT NULL DEFAULT (NOW() + INTERVAL '7 days'),
    accepted_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- =========================================================================
-- 7. CERTISWIFT AUTOMATION LOGISTICS
-- =========================================================================

CREATE TABLE certificate_templates (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    workspace_id UUID REFERENCES workspaces(id) ON DELETE CASCADE,
    title TEXT NOT NULL,
    base_image_url TEXT NOT NULL,
    signature_image_url TEXT,
    stamp_image_url TEXT,
    layout_coordinates JSONB NOT NULL,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE certificates (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    workspace_id UUID REFERENCES workspaces(id) ON DELETE CASCADE,
    template_id UUID REFERENCES certificate_templates(id) ON DELETE SET NULL,
    recipient_name TEXT NOT NULL,
    recipient_email TEXT NOT NULL,
    event_title TEXT NOT NULL,
    verification_hash TEXT UNIQUE NOT NULL, -- Public verification: elev8ed.app/verify/[hash]
    pdf_storage_url TEXT NOT NULL,
    issued_at TIMESTAMPTZ DEFAULT NOW()
);

-- =========================================================================
-- 8. THE FINANCES & SPONSORSHIP VAULT
-- =========================================================================

CREATE TABLE finances_and_sponsorships (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    workspace_id UUID REFERENCES workspaces(id) ON DELETE CASCADE,
    tenure_id UUID REFERENCES tenures(id) ON DELETE CASCADE,
    company_name TEXT NOT NULL,
    amount_secured NUMERIC(12, 2) NOT NULL DEFAULT 0.00,
    deal_status TEXT NOT NULL,
    contact_person TEXT,
    contact_email TEXT,
    deliverables_promised TEXT,
    notes TEXT,
    updated_by UUID REFERENCES members(id) ON DELETE SET NULL,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- =========================================================================
-- 9. PERFORMANCE LOOKUP INDEXES
-- =========================================================================

CREATE INDEX idx_workspaces_owner ON workspaces (owner_id);
CREATE INDEX idx_tasks_kanban_loop ON tasks (workspace_id, tenure_id, status);
CREATE INDEX idx_members_lookup ON members (workspace_id, tenure_id, status);
CREATE INDEX idx_archive_vault ON finances_and_sponsorships (workspace_id, tenure_id);
CREATE INDEX idx_certificates_verify ON certificates (verification_hash);
CREATE INDEX idx_invitations_token ON workspace_invitations (token);

-- =========================================================================
-- 10. ROW-LEVEL SECURITY (RLS) CORE GATEWAY POLICIES
-- =========================================================================

ALTER TABLE workspaces ENABLE ROW LEVEL SECURITY;
ALTER TABLE tenures ENABLE ROW LEVEL SECURITY;
ALTER TABLE members ENABLE ROW LEVEL SECURITY;

-- Workspaces: Owners can create; Owners and active members can view
CREATE POLICY "Users can create workspaces" ON workspaces FOR INSERT TO authenticated WITH CHECK (auth.uid() = owner_id);
CREATE POLICY "Users can view own workspaces" ON workspaces FOR SELECT TO authenticated USING (
    auth.uid() = owner_id OR id IN (SELECT workspace_id FROM members WHERE user_id = auth.uid())
);

-- Tenures: Workspace owners can create; Members and owners can view
CREATE POLICY "Owners can insert tenures" ON tenures FOR INSERT TO authenticated WITH CHECK (
    EXISTS (SELECT 1 FROM workspaces WHERE id = workspace_id AND owner_id = auth.uid())
);
CREATE POLICY "Members can view tenures" ON tenures FOR SELECT TO authenticated USING (
    EXISTS (SELECT 1 FROM workspaces WHERE id = workspace_id AND owner_id = auth.uid()) OR
    EXISTS (SELECT 1 FROM members WHERE workspace_id = tenures.workspace_id AND user_id = auth.uid())
);

-- Members: Founders can self-insert on setup; Users can view their own membership cards
CREATE POLICY "Owners can insert initial membership" ON members FOR INSERT TO authenticated WITH CHECK (
    auth.uid() = user_id OR EXISTS (SELECT 1 FROM workspaces WHERE id = workspace_id AND owner_id = auth.uid())
);
CREATE POLICY "Users can view own memberships" ON members FOR SELECT TO authenticated USING (user_id = auth.uid());

-- Departments:
-- 1. Enable RLS on the departments table
ALTER TABLE departments ENABLE ROW LEVEL SECURITY;

-- 2. Allow Workspace Owners to insert custom departments during setup
CREATE POLICY "Owners can insert departments" ON departments 
FOR INSERT TO authenticated 
WITH CHECK (
    EXISTS (SELECT 1 FROM workspaces WHERE id = workspace_id AND owner_id = auth.uid())
);

-- 3. Allow Owners and Active Members to view the departments on the dashboard
CREATE POLICY "Users can view departments" ON departments 
FOR SELECT TO authenticated 
USING (
    EXISTS (SELECT 1 FROM workspaces WHERE id = workspace_id AND owner_id = auth.uid()) OR
    EXISTS (SELECT 1 FROM members WHERE workspace_id = departments.workspace_id AND user_id = auth.uid())
);
